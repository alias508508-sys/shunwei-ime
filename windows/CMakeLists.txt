cmake_minimum_required(VERSION 3.20)
project(ShunweiIME VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(Shunwei SHARED src/tsf.cpp)
target_compile_definitions(Shunwei PRIVATE NOMINMAX UNICODE _UNICODE)
target_link_libraries(Shunwei PRIVATE ole32 oleaut32 uuid advapi32 user32 gdi32)
set_target_properties(Shunwei PROPERTIES PREFIX "")
add_executable(engine-check src/engine-check.cpp)
if(MSVC)
  target_compile_options(Shunwei PRIVATE /utf-8)
  target_compile_options(engine-check PRIVATE /utf-8)
  set_property(TARGET Shunwei engine-check PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    target_link_options(Shunwei PRIVATE "/EXPORT:DllCanUnloadNow=_DllCanUnloadNow@0" "/EXPORT:DllGetClassObject=_DllGetClassObject@12" "/EXPORT:DllRegisterServer=_DllRegisterServer@0" "/EXPORT:DllUnregisterServer=_DllUnregisterServer@0")
  else()
    target_link_options(Shunwei PRIVATE /EXPORT:DllCanUnloadNow /EXPORT:DllGetClassObject /EXPORT:DllRegisterServer /EXPORT:DllUnregisterServer)
  endif()
endif()
